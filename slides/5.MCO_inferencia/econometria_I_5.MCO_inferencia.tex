\begin{frame}
    \frametitle{Repaso: estimaci\'on por MCO}
    \begin{itemize}
        \bigskip
        \item \term{Consistencia}: $\plim \hat \beta = \beta$.
        \item \term{Distribuci\'on asint\'otica normal}: $\sqrt{N}(\hat \beta - \beta)\dto \N(0,V)$, donde $V=\E(x_ix_i')^{-1}\E(u_i^2x_ix_i')\E(x_ix_i')^{-1}$.
        \item \term{Estimaci\'on consistente de la matriz de varianzas y covarianzas}: $\hat V \pto V$.
        \item Usando estos resultados,
        \begin{align*}
            	&\hat \beta \adistr \N\left(\beta,\frac{V}{N}\right), \text{ y}\\
            	&s.e.(\hat \beta_j ) = \frac{\hat v_{jj}^{1/2}}{N^{1/2}} \quad \text{donde } v_{jj} \text{ es el elemento } jj \text{ de }V.
        \end{align*}
    \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Contraste de una restricci\'on lineal sobre un par\'ametro}
        \begin{itemize} 
            \item <1-> \term{Regla de decisi\'on}: Un estad\'istico $T_N$, un valor cr\'itico $c_\alpha$ y una regla de decisi\'on:
            \begin{enumerate}
                \item No rechazar $H_0$ si $T_N \leq c_\alpha$,
                \item Rechazar $H_0$ si $T_N >c_\alpha$.
            \end{enumerate}
            \item <2-> \term{Test estad\'istico}: Estad\'istico t en valor absoluto
            \begin{align*}
                |t_N|=\left|\frac{\hat \beta_j-\beta_{j,0}}{s.e.(\hat \beta_j)}  \right|
              \end{align*}
        \end{itemize}

\small
        \begin{itemize}
            \item \term{Test estad\'istico}: Estad\'istico t en valor absoluto
            \begin{align*}
                |t_N|=\left|\frac{\hat \beta_j-\beta_{j,0}}{s.e.(\hat \beta_j)} \right| \dto_{H_0} |Z|,
            \end{align*}
             donde $Z \sim \N(0,1)$ y $\Pr(|Z|\leq u)=2\Phi(u)-1$
            \item []  \term{Demostraci\'on}:
            \begin{align*}
                t_N&=\frac{\hat \beta_j-\beta_{j,0}}{s.e.(\hat \beta_j)}=\sqrt{N}\frac{\hat \beta_j-\beta_{j,0}}{\hat v_{jj}^{1/2}}\\
                   &=\sqrt{N}\frac{\hat \beta_j-\beta_{j,0}}{v_{jj}^{1/2}}\frac{v_{jj}^{1/2}}{\hat v_{jj}^{1/2}}\dto_{H_0} Z \sim \N(0,1),
            \end{align*}
            usando la distribuci\'on asint\'otica normal, $\hat v_{jj}\pto v_{jj}$ y el teorema de Slutsky.
        \end{itemize}
\end{frame}

	\begin{itemize}

		\begin{align*}
		|t_N|=\left|\frac{\hat \beta_j-\beta_{j,0}}{s.e.(\hat \beta_j)} \right| \dto_{H_0} |Z|
		\end{align*}
         donde $Z \sim \N(0,1)$ y $\Pr(|Z|\leq u)=2\Phi(u)-1$.
		\item []  \term{Demostraci\'on}: Como $t_N\dto_{H_0} Z \sim \N(0,1)$, por el teorema de la funci\'on continua $|t_N|\dto_{H_0} |Z|$. La probabilidad acumulada de $|Z|$ es
		\begin{align*}
		\Pr(|Z|\leq u)&=\Pr(-u\leq Z \leq u)= \Phi(u)-\Phi(-u)\\
		&=\Phi(u)-(1-\Phi(u))=2\Phi(u)-1.
		\end{align*}
	\end{itemize}
\end{frame}

\small
        \begin{itemize}
            \item \term{Test estad\'istico}: Estad\'istico t en valor absoluto
            \begin{align*}
            |t_N|=\left|\frac{\hat \beta_j-\beta_{j,0}}{s.e.(\hat \beta_j)} \right| \dto_{H_0} |Z|,
            \end{align*}
            donde $Z \sim \N(0,1)$ y $\Pr(|Z|\leq u)=2\Phi(u)-1$.
            \item  \term{Valor cr\'itico} $c_\alpha=z_{1-\alpha/2}$ donde  $\alpha$ es el tama\~no del test y $\Phi(z_{1-\alpha/2})=1-\alpha/2$.
            \item [] <2->  \term{Demostraci\'on}:
            \begin{align*}
                \lim_{N\to \infty}\Pr(\text{Rechazar }H_0|H_0)=\lim_{N\to \infty}\Pr(|t_N| >c_\alpha|H_0)= \Pr(|Z| >c_\alpha)=\alpha.
            \end{align*}
            Pero 
            \begin{align*}
            \Pr(|Z| >c_\alpha)&=1-\Pr(|Z| \leq c_\alpha)=1-(2\Phi(c_\alpha)-1)=2(1-\Phi(c_\alpha))=\alpha\\
            &\iff \Phi(c_\alpha)=1-\alpha/2 \iff c_\alpha=z_{1-\alpha/2}.
            \end{align*}
        \end{itemize}
\end{frame}

\begin{frame}
\frametitle{p-valor}
    El \term{p-valor} se define como $p_N=\Pr(T> T_N)$ donde $T_N\dto_{H_0} T$.
        \begin{itemize} 
            \item \term{Intuici\'on}: La regla de decisi\'on es rechazar $H_0$ si $T_N>c_\alpha$ donde $\Pr(T>c_\alpha)=\alpha$. Una regla de decisi\'on equivalente es calcular el p-valor $\Pr(T>T_N)=p_N$ y rechazar si $p_N<\alpha$.
            \item Dado que $\Pr(T>T_N)$ es decreciente en $T_N$, el p-valor  es una transformaci\'on mon\'otona de $T_N$ y por lo tanto es un estad\'istico equivalente. 
            \item \term{El p-valor es el nivel de significatividad marginal}: el menor valor de $\alpha$ para el cual se rechaza $H_0$.
            \item  Para $H_0: \beta_j = \beta_{j,0}$ y $H_1: \beta_j \neq \beta_{j,0}$ con el estad\'istico t tenemos:
            \begin{align*}
                p_N&=\Pr(T> T_N)=\Pr(|Z|>|t_N|)=2(1-\Phi(|t_N|)).
            \end{align*}
        \end{itemize}
\end{frame}

		\item El \term{poder del test} $\pi_N(\beta_j)$ es uno menos la probabilidad del error tipo II:
		\begin{align*}
    			\Pr(\text{Rechazar }H_0|H_1)&=\Pr(T_N >c_\alpha|H_1)=\pi_N(\beta_j).
    		\end{align*}
		\item <2- > En general, el poder del test aumenta a medida que $\beta_j$ se aleja de $\beta_{j,0}$ y a medida que el tama\~no de la muestra aumenta.
		\item <2- > No podemos reducir ambos errores a la vez: fijamos el tama\~no del test $\alpha$ y, entre los tests con ese tama\~no, preferimos el de mayor poder, que es el m\'as informativo sobre $H_0$.
		\item <2- > Si el poder es igual al tama\~no del test, el test no provee informaci\'on sobre la hip\'otesis de inter\'es.
	\end{itemize}
\end{frame}

\begin{frame}

    \begin{itemize}

        \item Para un nivel de confianza de $1-\alpha$:
        \begin{align*}
        \lim_{N\to \infty} \Pr\left(-c_\alpha<\frac{\hat \beta_j-\beta_j}{s.e.(\hat \beta_j)}< c_\alpha\right)&=\Pr\left(-c_\alpha<Z< c_\alpha\right)=1-\alpha
        \end{align*}

        \end{align*}

    \end{itemize}

\begin{frame}
\frametitle{?`Por qu\'e no usar dos tests t?}
        \begin{itemize}
            \item  Ejemplo: $H_0: \beta_1 = 0$ y $\beta_2 = 0$ vs. $H_1: \beta_1 \neq 0$ \'o $\beta_2 \neq 0$.
            \item  Regla: rechazar si $|t_{1,N}|>1.96$ \'o  $|t_{2,N}|>1.96$, donde $t_{k,N}=\hat \beta_k/s.e.(\hat \beta_k)$.
            \item  Supongamos que $\hat \beta_1$ y $\hat \beta_2$ son asint\'oticamente independientes. Entonces
             \begin{align*}
                \lim_{N\to\infty}\Pr_{H_0}(|t_{1,N}|>1.96 \text{ \'o } |t_{2,N}|>1.96)
                &=1-\lim_{N\to\infty}\Pr_{H_0}(|t_{1,N}|\leq 1.96 \text{ y } |t_{2,N}|\leq 1.96)\\
                &=1-\Pr(|Z_1|\leq 1.96) \times  \Pr(|Z_2|\leq 1.96)\\
                &= 1-(1-0.05)^2=0.0975.
            \end{align*}
            \item El tama\~no del test es \alert{9.75\%}, no 5\%. Si $\hat \beta_1$ y $\hat \beta_2$ est\'an correlacionados, el tama\~no depende de la correlaci\'on.
            \item Necesitamos un test conjunto de las dos restricciones.
        \end{itemize}
\end{frame}

        	\begin{align*}
        		\sqrt{N}(\hat \beta-\beta)&\dto \N(0,V),\\
        		\implies \sqrt{N}(R'\hat \beta-R'\beta)&\dto \N(0,R'VR),\\
        		\implies \sqrt{N}(R'\hat \beta-\theta_0)&\dto_{H_0} \N(0,R'VR).

        	Utilizando $\hat V\pto V$ y el teorema de Slutsky, bajo $H_0$ ($R'\beta=\theta_0$),
        		\begin{align*}
        			(R'\hat V R)^{-1/2}\sqrt{N}(R'\hat \beta-\theta_0)&\dto_{H_0} \N(0,I_q).
        		\end{align*}
        	Utilizando el teorema de la funci\'on continua y que $\sum_{k=1}^q z_k^2\sim \chi^2_q$ donde $z_k\sim\N(0,1)$ independientes,
        		\begin{align*}
        			W_N=N(R'\hat \beta-\theta_0)' (R'\hat V R)^{-1}(R'\hat \beta-\theta_0) \dto_{H_0} \chi^2_q.
        		\end{align*}
        \end{itemize}

    \frametitle{Relaci\'on entre el estad\'istico F y el estad\'istico de Wald}
    \begin{itemize}
        \item  Estad\'istico F:
        \begin{align*}
        F_N=\frac{(SSR_N^r-SSR_N^u)/q}{SSR_N^u/N}
        \end{align*}
        donde $SSR_N^u$ es la suma de los residuos al cuadrado del modelo no restringido y $SSR_N^r$ es la suma de los residuos al cuadrado del modelo restringido.
        \item Se cumple que
        \begin{align*}
        F_N=W_N/q
        \end{align*}
        \term{\'unicamente si $W_N$ utiliza el estimador de $V$ bajo homoscedasticidad}, $\hat V=\hat\sigma^2\left(\frac{1}{N}\sum_i x_ix_i'\right)^{-1}$. Con el estimador robusto a heteroscedasticidad no se cumple, y el estad\'istico $F_N$ basado en $SSR$ no es v\'alido bajo heteroscedasticidad.
    \end{itemize}
\end{frame}

        \item  \term{Test estad\'istico}: Estad\'istico de Wald
            \begin{align*}
            W_N=N\, (r(\hat \beta)-\theta_0)'(\hat R'\hat V \hat R)^{-1}(r(\hat \beta)-\theta_0) \dto_{H_0} \chi^2_q,
            \end{align*}
            donde $\hat R=\frac{\partial }{\partial \beta} r(\hat \beta)'$.
        \item  \term{Valor cr\'itico} $c_\alpha$:  $\Pr(\chi^2_q> c_\alpha)=\alpha$ \'o $\Pr(\chi^2_q\leq c_\alpha)=1-\alpha$.
        \item \term{Regla de decisi\'on}: Si $W_N>c_\alpha$, rechazamos $H_0$.
    \end{itemize}
\end{frame}

% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Restricciones no lineales}
    \begin{itemize}
        \item El test t y el test de Wald funcionan bien para hip\'otesis lineales pero tienen problemas para hip\'otesis no lineales 
        \item [] Ambos tests no son invariantes a c\'omo formulamos $H_0$.
        \item [] Ejemplo: $H_0:\beta_1=1$ y $H_0:1/\beta_1=1$ son la misma hip\'otesis, pero dan estad\'isticos de Wald distintos y pueden llevar a conclusiones distintas. Veremos un ejemplo con datos al final.
    \end{itemize}
\end{frame}
