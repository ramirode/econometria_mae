	\section{Modelo simple}
	\begin{frame}
		\frametitle{Modelo simple}
			\begin{align*}
			y_i=\beta_0 + \beta_1 x_{i}+u_i, \qquad \E(u_i)=0 \quad \text{(normalizaci\'on: $\beta_0$ absorbe la media de $u_i$)},
			\end{align*}
			donde $y_i$ es la variable dependiente, $x_i$ es una variable \term{end\'ogena} y $z_i$ es un \term{instrumento}.
			
			Supuestos: $\Cov(x_i,u_i) \neq 0$ ($x_i$ es end\'ogena)
			\begin{enumerate}
				\smallskip
				\item [IV1] Muestra aleatoria de tama\~no $N$ $\implies \{y_i,x_i,z_i\}_{i=1}^N$ i.i.d.,
				\smallskip
				\item [IV2]  \term{Exogeneidad del instrumento}: $\Cov(z_i,u_i)=0$,\\
				Dos condiciones: (1) $z_i$ es \emph{como si fuese} asignada en forma aleatoria,  (2) $z_i$ no tiene un efecto directo sobre $y_i$.
				\smallskip
				\item [IV3]  \term{Relevancia del instrumento}: $\Cov(z_i,x_i)\neq 0$
				\smallskip
				\item [IV4]  $\E(u_i^2)<\infty$ y $\E(z_i^2u_i^2)<\infty$.
			\end{enumerate}
	\end{frame}

	\subsection{Identificaci\'on}
	\begin{frame}
		\frametitle{Identificaci\'on}
		\begin{itemize}
			\item  ?`Qu\'e momentos poblacionales permiten identificar los par\'ametros de inter\'es?
			\item [] <2->
			\begin{align*}
				&\Cov(z_i,u_i)=0 \quad \text{usando IV2},\\
				&\iff \Cov(z_i,y_i-\beta_0-\beta_1x_i) =0,\\
				&\iff \Cov(z_i,y_i)-\beta_1\Cov(z_i,x_i) =0,\\
				&\iff \alert{\beta_1 =\frac{\Cov(z_i,y_i)}{\Cov(z_i,x_i)}}\quad \text{usando IV3}.
			\end{align*}
			\item  <3-> IV2 y IV3 son los supuestos que identifican a $\beta_1$.
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{M\'inimos Cuadrados Indirectos}
		\begin{itemize}
			\item \term{Primera etapa}
			\begin{align*}
				x_i=\pi_{10}+\pi_{11}z_i+\epsilon_{1i}.
			\end{align*}
			\item \term{Forma reducida}
			\begin{align*}
				y_i=\pi_{20}+\pi_{21}z_i+\epsilon_{2i}.
			\end{align*}
			\item  [] Entonces,
			\begin{align*}
				\beta_1=\frac{\Cov(z_i,y_i)}{\Cov(z_i,x_i)}=\frac{\Cov(z_i,y_i)/\Var(z_i)}{\Cov(z_i,x_i)/\Var(z_i)}
				\implies \alert{\beta_1=\frac{\pi_{21}}{\pi_{11}}}.
			\end{align*}
			\item Intuici\'on: Correlaci\'on entre $z_i$ e $y_i$ solamente se puede deber al efecto a trav\'es de $x_i$.
		\end{itemize}

	\begin{frame}
		\frametitle{M\'inimos Cuadrados en 2 Etapas (2SLS)}
		\vspace*{-0.6em}
		\begin{itemize}
			\item \term{Primera etapa}: $x_i=\pi_{10}+\pi_{11}z_i+\epsilon_{1i}=x_i^*+\epsilon_{1i}$.
			\item  Luego reemplazar $x_i$ por la predicci\'on $x_i^*$:
			\begin{align*}
				y_i&=\beta_{0}+\beta_1 x_i+u_{i},\\
				&=\beta_{0}+\beta_1x_i-\beta_1 x_i^*+\beta_1 x_i^*+u_{i},\\
				&=\beta_{0}+\beta_1 x_i^*+\beta_1(x_i-x_i^*)+u_{i}.
			\end{align*}
			\item [] $x_i^*=\pi_{10}+\pi_{11}z_i$ no est\'a correlacionada con $u_i$ (es funci\'on de $z_i$, IV2) ni con $x_i-x_i^*=\epsilon_{1i}$ (error de la proyecci\'on lineal, no correlacionado con $z_i$ por construcci\'on). Entonces,
			\begin{align*}
				\alert{ \beta_{1}=\frac{\Cov(x_i^*,y_i)}{\Var(x_i^*)} }.
			\end{align*}
			\item Intuici\'on: Solamente utilizamos la variaci\'on ex\'ogena de $x_i$.
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Caso especial: Estimador de Wald}
		\begin{itemize}
			\item  En el modelo
			\begin{align*}
				y_i=\beta_0 + \beta_1 x_{i}+u_i,
			\end{align*}
			con $z_i$ un instrumento v\'alido.
			\item  Supongamos que el instrumento es una variable binaria tal que
			\begin{align*}
				z_i=\left\{
				\begin{array}{c l}
					1 & \text{con probabilidad } p\\
					0 & \text{con probabilidad } 1-p
				\end{array}
				\right.
			\end{align*}
			\item  [] Entonces el par\'ametro IV es igual al llamado estimador de Wald:
			\begin{align*}
				\beta_{1, Wald}=\frac{\E(y|z=1)-\E(y|z=0)}{\E(x|z=1)-\E(x|z=0)}.
			\end{align*}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Estimador de Wald}
		\begin{itemize}
			\item  \alert{El estimador de Wald es un estimador IV}: $\beta_{1,Wald}=\beta_{1,IV}$
			\item  [] Demostraci\'on: Utilice la LEI para escribir
			\small
			\begin{align*}
				\E(y) = (1-p) \E(y|z=0)+ p \E(y|z=1), \qquad \E(zy) = \E[z\E(y|z)]= p \E(y|z=1).
			\end{align*}
			Luego reemplace en la definici\'on de la covarianza
			\begin{align*}
				\Cov(z,y) &= \E(zy) - \E(z)\E(y) \\
				&= p \E(y|z=1) - p[(1-p) \E(y|z=0)+ p \E(y|z=1)] \\
			 	&= p(1-p)[\E(y|z=1)-\E(y|z=0)],
			\end{align*}
			y en forma an\'aloga $\Cov(z,x)=p(1-p)[\E(x|z=1)-\E(x|z=0)]$. Finalmente reemplace en la definici\'on de $\beta_{1,IV}$
			\begin{align*}
				\beta_{1,IV}=\frac{\Cov(z,y)}{\Cov(z,x)}=\frac{\E(y|z=1)-\E(y|z=0)}{\E(x|z=1)-\E(x|z=0)}=\beta_{1,Wald}.
			\end{align*}
		\end{itemize}
	\end{frame}

	\subsection{Estimaci\'on}
	\begin{frame}
		\frametitle{Estimaci\'on}
		\begin{itemize}
			\item  \term{Estimador por variables instrumentales (IV)}: utilizamos el principio de analog\'ia para reemplazar momentos poblacionales por momentos muestrales.
			\item  [] Estamos interesados en:
			\begin{align*}
				\beta_1 =\frac{\Cov(z_i,y_i)}{\Cov(z_i,x_i)}.
			\end{align*}
			\uncover <2-> {
			Usando el principio de analog\'ia, lo estimamos con
			\begin{align*}
				\hat \beta_{1,IV} =\frac{\widehat{\Cov}(z_i,y_i)}{\widehat{\Cov}(z_i,x_i)}.
			\end{align*}
			}
		\end{itemize}
	\end{frame}
	% --------------------------------------------------- Slide --
    \begin{frame}
        \frametitle{Estimaci\'on}
        \begin{itemize}
            \item \term{Estimador por variables instrumentales (IV)}: 
            Usando el principio de analog\'ia, lo estimamos con        
            \begin{align*}
                \hat \beta_{1,IV} =\frac{\widehat{\Cov}(z_i,y_i)}{\widehat{\Cov}(z_i,x_i)}.
            \end{align*}          
            \item <2-> En forma similar podemos motivar el \term{estimador por M\'inimos Cuadrados Indirectos}         
            \begin{align*}
                \hat \beta_{1, \text{ILS}}&=\frac{\hat\pi_{21}}{\hat \pi_{11}},
            \end{align*}          
            \item [] <3-> y el \term{estimador 2SLS}
            \begin{align*}
                \hat \beta_{1, \text{2SLS}}&=\frac{\widehat{\Cov}(\hat x_i,y_i)}{\widehat{\Var}(\hat x_i)},
                \quad \text{donde } \hat x_i=\hat \pi_{10}+\hat\pi_{11}z_i.
            \end{align*}
        \end{itemize}
    \end{frame}

	\begin{frame}
		\frametitle{Los estimadores son equivalentes: $\hat \beta_{1,IV}=\hat \beta_{1,ILS}=\hat \beta_{1,\text{2SLS}}$}
        	Demostraci\'on:
		\begin{itemize}
            		\item $\hat \beta_{1,IV}=\hat \beta_{1,ILS}$
			\begin{align*}
				\hat \beta_{1,IV} =\frac{\widehat{\Cov}(z_i,y_i)}{\widehat{\Cov}(z_i,x_i)}=\frac{\widehat{\Cov}(z_i,y_i)/\widehat{\Var}( z_i)}{\widehat{\Cov}(z_i,x_i)/\widehat{\Var}( z_i)}=\frac{\hat\pi_{21}}{\hat \pi_{11}}=\hat \beta_{1,ILS}.
			\end{align*}
			donde $x_i=\hat \pi_{10}+\hat \pi_{11}z_i+\hat \epsilon_{1i}$ y $y_i=\hat \pi_{20}+\hat \pi_{21}z_i+\hat \epsilon_{2i}$.
			\item  $\hat \beta_{1,2SLS}=\hat \beta_{1,IV}$
			\begin{align*}
				\hat \beta_{1, \text{2SLS}}&=\frac{\widehat{\Cov}(\hat x_i,y_i)}{\widehat{\Var}(\hat x_i)}
				=\frac{\widehat{\Cov}(\hat x_i,y_i)}{\widehat{\Cov}(\hat x_i,x_i)}
				=\frac{\hat\pi_{11}\widehat{\Cov}(z_i,y_i)}{\hat\pi_{11}\widehat{\Cov}(z_i,x_i)}=\hat \beta_{1,IV}.
			\end{align*}
			donde $\hat x_i=\hat \pi_{10}+\hat\pi_{11}z_i$, y $\widehat{\Var}(\hat x_i)=\widehat{\Cov}(\hat x_i,x_i)$ porque $x_i=\hat x_i+\hat\epsilon_{1i}$ y $\hat x_i$ no est\'a correlacionada con el residuo de la primera etapa: $\widehat{\Cov}(\hat x_i,\hat \epsilon_{1i})=0$.
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Retornos de la educaci\'on (Angrist y Krueger, QJE 1991)}
		\begin{itemize}
			\item  ?`Cu\'ales son los retornos de la educaci\'on?
			\item Estamos interesados en el modelo
			\begin{align*}
				y_i=\alpha+\rho s_i+\gamma a_i+\nu_i
			\end{align*}
			donde $y_i$ es log de ingresos, $s_i$ es escolaridad, $a_i$ es habilidad innata y $\nu_i$ son inobservados que cumplen $\Cov(s_i,\nu_i)=\Cov(a_i,\nu_i)=0$.
			\item Si observamos habilidad innata, podemos estimar el efecto causal por MCO.
			\item Desafortunadamente, no la observamos: en el modelo que podemos estimar, $y_i=\alpha+\rho s_i+u_i$ con $u_i=\gamma a_i+\nu_i$, MCO tiene sesgo por variable omitida.
			\item  Soluci\'on de variables instrumentales: una variable correlacionada con $s_i$ pero que no est\'e correlacionada con $u_i$.
		\end{itemize}
	\end{frame}
	\begin{frame}
		\frametitle{Retornos de la educaci\'on (Angrist y Krueger, QJE 1991)}

\subsection{Propiedades}
\begin{frame}
    \frametitle{Propiedades de $\hat \beta_{IV}$}
    \vspace*{-0.6em}
    \small
    \begin{itemize}
        \item  <1->Hasta ahora demostramos que, en el modelo de regresi\'on simple con un instrumento, los estimadores IV, ILS y 2SLS son id\'enticos.
        \item  <1-> Adem\'as, el estimador de Wald es un caso particular de IV cuando el instrumento es una variable dummy.
        \item  <2->Dado que los estimadores son id\'enticos, analizamos las propiedades del estimador IV:
        \begin{align*}
            \hat \beta_{1,IV} =\frac{\widehat{\Cov}(z_i,y_i)}{\widehat{\Cov}(z_i,x_i)},
        \end{align*}
        con $\E(u_i)=0$, bajo los supuestos de
    \begin{enumerate}\small
        \item [IV1]  Muestra aleatoria de tama\~no $N$ $\implies \{y_i,x_i,z_i\}_{i=1}^N$ i.i.d.,
        \item [IV2]  Exogeneidad del instrumento: $\Cov(z_i,u_i)=0$,
        \item [IV3]  Relevancia del instrumento: $\Cov(z_i,x_i)\neq 0$
        \item [IV4]  $\E(u_i^2)<\infty$ y $\E(z_i^2u_i^2)<\infty$ (para la normalidad asint\'otica).
    \end{enumerate}
        \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Consistencia}
        \begin{align*}
            \hat \beta_{1,IV} &=\frac{\widehat{\Cov}(z_i,y_i)}{\widehat{\Cov}(z_i,x_i)}= \frac{\widehat{\Cov}(z_i,\beta_0+\beta_1 x_i+u_i)}{\widehat{\Cov}(z_i,x_i)},\\
            & = \beta_1+\frac{\widehat{\Cov}(z_i,u_i)}{\widehat{\Cov}(z_i,x_i)},
        \end{align*}
        usando los resultados derivados en el cap\'itulo de Teor\'ia asint\'otica, y IV2 (exogeneidad) y IV3 (relevancia)
        \begin{align*}
            \plim \widehat{\Cov}(z_i,x_i) & = {\Cov}(z_i,x_i) \neq 0, \\
            \plim \widehat{\Cov}(z_i,u_i) & = {\Cov}(z_i,u_i) = 0.
        \end{align*}
        Entonces    $\plim \hat \beta_{1,IV}=\beta_{1}$.
\end{frame}

\begin{frame}
    \frametitle{Distribuci\'on asint\'otica normal}
    \vspace*{-0.6em}
        \begin{align*}
            \sqrt{N} (\hat \beta_{1,IV} - \beta_1)&= \frac{1}{\widehat{\Cov}(z_i,x_i)}\sqrt{N}\widehat{\Cov}(z_i,u_i).
        \end{align*}
        Usando los resultados del cap\'itulo de Teor\'ia asint\'otica, IV2, IV3 y IV4,
        \begin{align*}
            \plim \widehat{\Cov}(z_i,x_i) & = {\Cov}(z_i,x_i) \neq 0, \\
            \sqrt{N}\widehat{\Cov}(z_i,u_i) &\dto \N (0,V).
        \end{align*}
        donde 
        \begin{align*}
            V&=\E[(z-\E z)^2u^2]-\E[(z-\E z)u]^2,\\
            &=\E[(z-\E z)^2u^2]-\Cov(z,u)^2,\\
            &=\E[(z-\E z)^2u^2].
        \end{align*}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Distribuci\'on asint\'otica normal}
    \begin{itemize}
        \item [] Usando el teorema de Slutsky
        \begin{align*}
            \sqrt{N}(\hat \beta_{1,IV} - \beta_1) &\dto \N \left(0,\frac{\E[(z-\E z)^2u^2]}{\Cov(z,x)^2}\right).
        \end{align*}
        \item  \alert{Solamente} para obtener intuici\'on y comparar con MCO asumimos homoscedasticidad ($\E(u^2|z)=\E(u^2)=\sigma^2$) y obtenemos:
        \item [] 
        \begin{align*}
        	V&=\E[(z-\E z)^2u^2]=\Var(z)\E(u^2),\\
            \sqrt{N}(\hat \beta_{1,IV} - \beta_1) &\dto \N \left(0,\frac{\Var(z)\sigma^2}{\Cov(z,x)^2}\right)\\
            &= \N \left(0,\frac{\sigma^2}{\rho_{z,x}^2\Var(x)}\right).
        \end{align*}
    \end{itemize}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Distribuci\'on asint\'otica normal}
    \begin{itemize}
        \item  Decimos
                \begin{align*}
                    &\hat \beta_{1,IV} \adistr \N(\beta_1,\Avar(\hat\beta_{1,IV})),\text{ donde}\\
                    &\Avar(\hat \beta_{1,IV})=\frac{1}{N}\frac{\E[(z-\E z)^2u^2]}{\Cov(z,x)^2}.
                \end{align*}
        \item Bajo homoscedasticidad,
                \begin{align*}
                    \Avar(\hat \beta_{1,IV})=\frac{1}{N}\frac{\sigma^2}{\rho_{z,x}^2\Var(x)}.
                \end{align*}
    \end{itemize}
\end{frame}

% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Comparaci\'on de la varianza asint\'otica de los estimadores MCO e IV}
    Bajo homoscedasticidad:
            \begin{align*}
            \Avar(\hat \beta_{1,OLS})&=\frac{\sigma^2}{N}\frac{1}{\Var(x)},\\
            \Avar(\hat \beta_{1,IV})&=\frac{\sigma^2}{N}\frac{1}{\rho_{z,x}^2\Var(x)}.
            \end{align*}
\end{frame}

\subsection{Instrumentos d\'ebiles}
% --------------------------------------------------- Slide --
\begin{frame}
	\frametitle{Instrumentos d\'ebiles}
    \begin{itemize}
        \item  \term{Instrumento d\'ebil}: $z$ explica poco de la variaci\'on de $x$ ($\rho_{z,x}$ cercano a cero).
        \item \term{Consecuencias}:
        \begin{itemize}
            \item Si $z$ est\'a algo correlacionado con $u$, el sesgo asint\'otico de IV puede ser mayor que el de MCO: un $\rho_{z,x}$ peque\~no amplifica cualquier violaci\'on de la exogeneidad.
            \item En muestras finitas, 2SLS est\'a sesgado hacia MCO. El sesgo es mayor cuanto m\'as d\'ebiles son los instrumentos y cuantos m\'as instrumentos se usan.
            \item La distribuci\'on de $\hat \beta_{1,IV}$ no se parece a una normal: los errores est\'andar no reflejan la incertidumbre, los tests $t$ rechazan $H_0$ verdadera con demasiada frecuencia, y los intervalos de confianza contienen $\beta_1$ con menor probabilidad que la nominal.
        \end{itemize}
    \end{itemize}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Sesgo asint\'otico cuando el instrumento no es ex\'ogeno}
            \begin{align*}
            \plim \hat \beta_{1,OLS} &= \beta_1+\frac{\Cov(x,u)}{\Var(x)}=\beta_1+\frac{\sigma_u}{\sigma_x}\rho_{x,u},\\
            \plim \hat \beta_{1,IV} &= \beta_1+\frac{\Cov(z,u)}{\Cov(z,x)}=\beta_1+\frac{\sigma_u}{\sigma_x}\frac{\rho_{z,u}}{\rho_{z,x}},\\
            \text{sesgo}(\hat \beta_{1,IV}) &\leq \text{sesgo}(\hat \beta_{1,OLS}),\\
            \iff \frac{\rho_{z,u}}{\rho_{z,x}}& \leq \rho_{x,u}. 
            \end{align*}
\end{frame}

\begin{frame}
	\frametitle{Simulaci\'on: MCO e IV en muestras finitas}
    \begin{itemize}
    	\item  Modelo generador de datos
    	\begin{align*}
    	y_i = 1+ x_i+u_i,
    	\end{align*}
    	y tenemos un instrumento disponible $z_i$ tal que
    	\begin{align*}
    	\begin{pmatrix}
    	u \\
    	z \\
    	x
    	\end{pmatrix}
    	\sim\N
    	\left(
    	0,
    	\begin{pmatrix}
    	1 & \rho_{z,u} & \rho_{x,u} \\
    	\rho_{z,u} & 1 & \rho_{z,x} \\
    	\rho_{x,u} & \rho_{z,x} & 1
    	\end{pmatrix}
    	\right)
    	\end{align*}
    	\item $\rho_{x,u}$: exogeneidad de $x$; $\rho_{z,u}$: exogeneidad de $z$; $\rho_{z,x}$: relevancia de $z$.
    	\item Tenemos una muestra aleatoria de tama\~no $N=500$, $\{y_i, x_i,z_i\}_{i=1}^{500}$, y repetimos la simulaci\'on 500 veces.
    \end{itemize}

\begin{frame}
    \frametitle{?`C\'omo detectar instrumentos d\'ebiles? El $F$ de la primera etapa}
    \begin{itemize}
        \item Primera etapa $x_i=\pi_{10}+\pi_{11}z_i+\epsilon_{1i}$: contrastar $H_0:\pi_{11}=0$ (con varios instrumentos, test $F$ conjunto de sus coeficientes).
        \item \term{Regla de Staiger y Stock (1997)}: $F>10$. Se dise\~n\'o con errores homosced\'asticos, para que el sesgo de 2SLS sea peque\~no en relaci\'on al de MCO.
        \item Dos problemas con esta regla:
        \begin{itemize}
            \item Con heteroscedasticidad hay que usar el $F$ robusto (el \term{$F$ efectivo} de Montiel Olea y Pflueger, 2013; con un instrumento es igual al $F$ robusto).
            \item $F>10$ no garantiza que el test $t$ al 5\% sea v\'alido. Con un instrumento, se necesita $F>104.7$ para usar $\pm 1.96$ (Lee, McCrary, Moreira y Porter, AER 2022).
        \end{itemize}
        \item El $F$ sirve para diagnosticar, pero no para ``aprobar'' la inferencia habitual de 2SLS.
    \end{itemize}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Test de Anderson--Rubin}
    \begin{itemize}
        \item Queremos contrastar $H_0:\beta_1=b$. Definimos $\tilde y_i=y_i-b\,x_i$:
        \begin{align*}
            \tilde y_i=\beta_0+(\beta_1-b)\,x_i+u_i.
        \end{align*}
        \item Usando IV2, $\Cov(z_i,\tilde y_i)=(\beta_1-b)\Cov(z_i,x_i)$. Bajo $H_0$, $\Cov(z_i,\tilde y_i)=0$ \emph{sea cual sea} $\Cov(z_i,x_i)$.
        \item \term{Test de Anderson--Rubin (AR)}: regresar $\tilde y_i$ sobre $z_i$ (y las ex\'ogenas incluidas) por MCO, y contrastar con un test $t$ o $F$ robusto que el coeficiente de $z_i$ es cero.
        \item En el modelo simple, el coeficiente de $z_i$ es $\pi_{21}-b\,\pi_{11}$ (forma reducida menos $b$ veces la primera etapa). No se divide por $\hat \pi_{11}$: por eso el test es v\'alido aunque $\pi_{11}$ sea peque\~no.
    \end{itemize}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Intervalo de confianza de Anderson--Rubin}
    \begin{itemize}
        \item \term{Invertir el test}: el intervalo de confianza al 95\% es el conjunto de valores $b$ que el test AR no rechaza al 5\%.
        \item En la pr\'actica: una grilla de valores de $b$ y un test AR para cada uno.
        \item Con un instrumento d\'ebil, el conjunto puede ser muy ancho o incluso no acotado. Es la respuesta correcta: los datos son poco informativos sobre $\beta_1$.
        \item Con una variable end\'ogena y un instrumento, es el procedimiento recomendado (Andrews, Stock y Sun, 2019; Keane y Neal, 2024). Con m\'as instrumentos sigue siendo v\'alido, pero puede tener menos poder que otras alternativas.
    \end{itemize}
\end{frame}

    \vspace*{-0.6em}
    \small
    \begin{align*}
        y_i=x_{i}'\beta +u_i,
    \end{align*}
    donde 
    \begin{itemize}
        \item $x_i$ son $K$ \term{variables explicativas} (incluida la constante) de donde $K_1$ son \term{ex\'ogenas} y $K_2$ son \term{end\'ogenas} ($K=K_1+K_2$), y 
        \item $z_i$ son  $L$ \term{variables ex\'ogenas} donde $K_1$ son las variables \term{ex\'ogenas incluidas} en el modelo y $L_1$ son variables \term{ex\'ogenas excluidas} en el modelo (\term{instrumentos}) ($L=K_1+L_1$).
    \end{itemize}
    \uncover <2-> {
    Suponemos que $L=dim(z)\geq dim(x)=K \iff L_1\geq K_2$: Tenemos al menos tantos instrumentos como variables end\'ogenas.
   
    Si $L=K$ (y se cumple una condici\'on de rango) el modelo est\'a \term{exactamente identificado}; si $L>K$ el modelo est\'a potencialmente \term{sobreidentificado}.
    }
\end{frame}

    \end{enumerate}

    \begin{itemize} 
        \item La condici\'on $rango(\E(z x'))=K$ es la generalizaci\'on del supuesto de relevancia del modelo simple.

        \item[] Entonces:
        \begin{align*}
        \E(zx')=\left(
        \begin{array}{c c}
        1 & \E(x_1)\\

        \end{array}
        \right).
        \end{align*}
        $rango(\E(zx'))=2$ si y s\'olo si $|\E(zx')|\neq 0$. 
        \item[] <2-> $|\E(zx')|=\E(z_1 x_1)-\E(z_1)\E(x_1)=\Cov(z_1, x_1)\neq 0$.
    \end{itemize}

\begin{frame}
	\frametitle{Condici\'on de Rango}
	\begin{itemize} 
		\item  Para \term{una variable end\'ogena y un instrumento}, podemos contrastar la condici\'on de rango con un test de significancia del instrumento en la primera etapa.
		\item Para \term{una variable end\'ogena y m\'ultiples  instrumentos}, podemos contrastar la condici\'on de rango con un test de significancia conjunta de los instrumentos en la primera etapa.
		\item Para \term{m\'as de una variable end\'ogena}, una forma intuitiva de verla es que instrumentos distintos sean significativos para distintas variables end\'ogenas en la primera etapa. Existen contrastes que eval\'uan si la matriz $\E(zx')$ tiene rango $K$.
		\item  Contraste de Wald (o F) en la primera etapa $H_0:$ los coeficientes de los instrumentos son cero, versus $H_1:$ al menos uno de los coeficientes de los instrumentos es distinto de cero.
	\end{itemize}

\subsection{Identificaci\'on y Estimaci\'on}
\begin{frame}
    \frametitle{Identificaci\'on para $L=K$}
    \begin{itemize}
        \item ?`Qu\'e momentos poblacionales permiten identificar los par\'ametros de inter\'es?
        \item [] <2->
        \begin{align*}
        		&\E(z u)=0 \quad \text{usando IV2},\\
        		&\iff \E(z (y-x'\beta)) =0,\\
        		&\iff \E(z y)-\E(z x')\beta =0,\\
        		&\iff \beta=\E(z x')^{-1}\E(z y)\quad \text{usando IV4}.
        \end{align*}
        \item  <2-> IV2 y IV4 son los supuestos que identifican $\beta$.
    \end{itemize}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Estimador por variables instrumentales (IV)}
    \begin{itemize}
        \item  Principio de analog\'ia: reemplazar momentos poblacionales por momentos muestrales.
        \item [] <2-> Estamos interesados en:
        \begin{align*}
        		\beta =\E(z x')^{-1}\E(z y).
        \end{align*}
        Usando el principio de analog\'ia, lo estimamos con
        \begin{align*}
        		\hat \beta_{IV} 
        		=\left(
        		\frac{1}{N} \sum z_i x_i'
        		\right)^{-1}\frac{1}{N}\sum z_i y_i .
        \end{align*}
        \item <2-> En notaci\'on matricial: $\hat \beta_{IV}=(Z'X)^{-1}Z'Y$ donde $Z=(z_1,...,z_N)'$.
    \end{itemize}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
	\frametitle{$L>K$: ?`Por qu\'e no aplicar directamente el estimador IV?}
	\begin{itemize}
		\item Con $L=K$, los momentos muestrales $\frac{1}{N}\sum_i z_i(y_i-x_i'b)=0$ son $K$ ecuaciones con $K$ inc\'ognitas, y su soluci\'on es
		\begin{align*}
			\hat\beta_{IV}=\left(\frac{1}{N}\sum_i z_ix_i'\right)^{-1}\frac{1}{N}\sum_i z_iy_i.
		\end{align*}
		\item Con $L>K$ son $L$ ecuaciones con $K$ inc\'ognitas: en general ning\'un $b$ las cumple todas en la muestra, y $\frac{1}{N}\sum_i z_ix_i'$ es de $L\times K$, no se puede invertir.
		\item \term{Soluci\'on}: usar $K$ combinaciones lineales de los instrumentos, $A'z_i$, con $A$ de $L\times K$ y rango $K$. Como $\E(A'zu)=A'\E(zu)=0$, \emph{cualquier} $A$ da un estimador IV consistente.
		\item Ejemplo (Graddy): con cuatro instrumentos, podemos usar solo $wave2$, solo $speed3$, o cualquier combinaci\'on de los cuatro. ?`Cu\'al conviene?
	\end{itemize}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
	\frametitle{?`Qu\'e combinaci\'on usar? La primera etapa}
	\small
	\begin{itemize}
		\item Se puede demostrar que, bajo homoscedasticidad, la mejor combinaci\'on lineal de los instrumentos es la que da la \term{primera etapa}: la proyecci\'on lineal de $x$ sobre $z$.
		\item Primera etapa, una ecuaci\'on por cada variable explicativa:
		\begin{align*}
		x=\Pi'z+e,\qquad \Pi=\E(zz')^{-1}\E(zx') \text{ de } L\times K, \qquad \E(ze')=0.
		\end{align*}
		\item El vector de instrumentos de $x$ es $x^*=\Pi'z$: es decir, $A=\Pi$.
		\item Cada variable ex\'ogena incluida $x_j$ es parte de $z$, as\'i que su primera etapa la predice exactamente: $x_j^*=x_j$. Solo se reemplazan las end\'ogenas.
		\item Graddy: $x^*=(1,\, mon,\, tues,\, wed,\, thurs,\, lavgprc^*)$, donde $lavgprc^*$ es la proyecci\'on lineal de $lavgprc$ sobre los d\'ias de la semana, $speed2$, $speed3$, $wave2$ y $wave3$.
	\end{itemize}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
	\frametitle{La primera etapa en forma matricial}
	\small
	\begin{itemize}
		\item Para cada variable explicativa $x_j$, la proyecci\'on lineal sobre $z$ es
		\begin{align*}
		x_j=\pi_j'z+e_j,\qquad \pi_j=\E(zz')^{-1}\E(zx_j),\qquad \E(ze_j)=0, \qquad j=1,\dots,K.
		\end{align*}
		\item Apilando las $K$ ecuaciones:
		\begin{align*}
		x = \left[
		\begin{array}{c}
		\pi_1'\\
		\vdots\\
		\pi_K'
		\end{array}
		\right]z+e=\Pi'z+e, \qquad \Pi=[\pi_1 \ldots \pi_K],\qquad \E(ze')=0.
		\end{align*}
		\item Entonces
		\begin{align*}
		\Pi=[\pi_1 \ldots \pi_K]=\E(zz')^{-1} [\E(zx_1) \ldots \E(zx_K)]=\E(zz')^{-1} \E(zx').
		\end{align*}
	\end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Instrumentos}
	Queremos hacer una estimaci\'on de IV con $x^*=\Pi'z$. Verifiquemos que los instrumentos cumplen la condici\'on de exogeneidad y rango.
    \begin{itemize}
    \item \term{Exogeneidad}
    \item []  $\E(x^*u)=\E(\Pi'zu)=\Pi'\E(zu)=0$ por exogeneidad de $z$.
    \item \term{Condici\'on de rango}: $rango(\E(x^*x'))=K$.
    \item []  $\E(x^*x')=\E(\Pi'zx')=\Pi'\E(zx')=\E(xz')\E(zz')^{-1}\E(zx')$\\
    Como $rango(\E(zz'))=L$, $rango(\E(zx'))=K$ y $L \geq K$ entonces $rango(\E(x^*x'))=K$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Estimaci\'on}
    Estimaci\'on en dos etapas:
    \begin{itemize}
    \item Regresar $x_i$ en $z_i$ y obtener valores predichos 
    \begin{align*}
    \hat x_i = \hat\Pi' z_i.
            \end{align*}
    \item [] Nota: Es equivalente a regresar solamente las variables end\'ogenas en $z_i$, porque cada variable ex\'ogena incluida es su propio instrumento ($\hat x_{ji}=x_{ji}$).
    \item Usar $\hat x_i$ como instrumentos en un IV
            \begin{align*}
                \hat \beta_{IV} 
                =\left(
                \frac{1}{N} \sum \hat x_i x_i'
                \right)^{-1}\frac{1}{N}\sum \hat x_i y_i .
            \end{align*}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Estimaci\'on}
    \vspace*{-0.6em}
    \small
    \begin{itemize}
    \item Dado que $x_i = \hat x_i + \hat e_i$ y de las CPO de MCO
        \begin{align*}
        \frac{1}{N}\sum_i \hat x_i \hat e_i'=0,
        \end{align*}
        entonces
        \begin{align*}
        \frac{1}{N}\sum_i \hat x_i x_i'=\frac{1}{N}\sum_i \hat x_i \hat x_i',
        \end{align*}
        y el estimador IV se puede escribir como un estimador 2SLS
            \begin{align*}
                \hat \beta_{2SLS} 
                =\left(
                \frac{1}{N} \sum \hat x_i \hat x_i'
                \right)^{-1}\frac{1}{N}\sum \hat x_i y_i,
            \end{align*}
            donde la segunda etapa es una regresi\'on de $y$ en $\hat x_i$.
            \item Recuerde: No es recomendable computar el estimador usando el procedimiento en dos etapas porque los s.e. est\'an mal calculados.
    \end{itemize}
\end{frame}
\subsection{Propiedades}
\begin{frame}
    \frametitle{Propiedades}
    \small
    \begin{itemize} 
    	\item Para probar consistencia es conveniente escribir el estimador en funci\'on de las variables originales.
    	\item El coeficiente estimado de regresar $x_j$ en los instrumentos es  $\hat \pi_j=\left(\sum z_i z_i'\right)^{-1} \sum z_i x_{ji}$.
    	\item La matriz de dichos coeficientes es $\hat \Pi=[\hat \pi_1 \hat \pi_2 ... \hat \pi_K]=\left(\sum z_i z_i'\right)^{-1} \sum z_i x_{i}'$.
    	\item Por lo tanto podemos escribir:
        \begin{align*}
            \hat \beta_{2SLS} &=\left(
                            \sum \hat x_i \hat x_i'
                            \right)^{-1}\sum \hat x_i y_i,\\
            &=\left[\left(\sum x_i z_i'\right)\left(\sum z_i z_i'\right)^{-1}\left(\sum z_i x_i'\right) \right]^{-1}\\
            &
            \left(\sum x_i z_i'\right)\left(\sum z_i z_i'\right)^{-1}\left(\sum z_i y_i\right).
        \end{align*}
 %       \item En notaci\'on matricial: $\hat \beta_{2SLS}=[(X'Z)(Z'Z)^{-1}(Z'X)]^{-1}(X'Z)(Z'Z)^{-1}(Z'Y)$.
        \item Si $K=L$, tenemos el estimador IV.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Consistencia}
        \begin{align*}
            \hat \beta_{2SLS} &=\beta+\left[\left(\frac{1}{N}\sum x_i z_i'\right)\left(\frac{1}{N}\sum z_i z_i'\right)^{-1}\left(\frac{1}{N}\sum z_i x_i'\right) \right]^{-1}\\
                        &
                        \left(\frac{1}{N}\sum x_i z_i'\right)\left(\frac{1}{N}\sum z_i z_i'\right)^{-1}\left(\frac{1}{N}\sum z_i u_i\right)
        \end{align*}
        Entonces usando los resultados derivados en el cap\'itulo de Teor\'ia asint\'otica, y IV2, IV3 y IV4
        \begin{align*}
            \plim\hat \beta_{2SLS} &=\beta+[\E(xz')\E(zz')^{-1}\E(zx')]^{-1}\E(xz')\E(zz')^{-1}\E(zu).
        \end{align*}
        Entonces $\plim \hat \beta_{2SLS}=\beta$.
\end{frame}

\begin{frame}
    \frametitle{Distribuci\'on asint\'otica normal}
    \vspace*{-0.6em}
    \small
        \begin{align*}
            \sqrt{N} (\hat \beta_{2SLS} - \beta)&= \left[\left(\frac{1}{N}\sum x_i z_i'\right)\left(\frac{1}{N}\sum z_i z_i'\right)^{-1}\left(\frac{1}{N}\sum z_i x_i'\right) \right]^{-1}\\
                                    &
                                    \left(\frac{1}{N}\sum x_i z_i'\right)\left(\frac{1}{N}\sum z_i z_i'\right)^{-1}\frac{1}{\sqrt{N}} \sum z_i u_i
        \end{align*}
        Usando los resultados derivados en el cap\'itulo de Teor\'ia asint\'otica, y IV2, IV3, IV4 y IV5, y el  teorema de Slutsky,
        \begin{align*}
            \sqrt{N} (\hat \beta_{2SLS} - \beta)&\dto \N(0,B^{-1}CB^{-1}),
        \end{align*}
        donde 
        \begin{align*}
            B&=\E(xz')\E(zz')^{-1}\E(zx'),\\
            C&=\E(xz')\E(zz')^{-1}\E(u^2zz') \E(zz')^{-1}\E(zx').
        \end{align*}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Distribuci\'on asint\'otica normal}
    \vspace*{-0.6em}
    \small
    \begin{itemize}
        \item  Decimos
                \begin{align*}
                    &\hat \beta_{2SLS} \adistr \N(\beta,\Avar(\hat\beta_{2SLS})),\text{ donde}\\
                    &\Avar(\hat \beta_{2SLS})=\frac{1}{N}B^{-1}CB^{-1}.
                \end{align*}
         \item  Estimaci\'on consistente de $\Avar(\hat \beta_{2SLS})$: Reemplazamos momentos poblacionales por muestrales y errores poblacionales ($u_i$) por residuos ($\hat u_i$). Es decir,
                \begin{align*}
                    \widehat{\Avar}(\hat \beta_{2SLS})=\frac{1}{N}\hat B^{-1}\hat C \hat B^{-1},
                \end{align*}
                donde
                \scriptsize
                \begin{align*}
                    \hat B=&\left(\frac{1}{N}\sum x_i z_i'\right)\left(\frac{1}{N}\sum z_i z_i'\right)^{-1}\left(\frac{1}{N}\sum z_i x_i'\right),\\
                    \hat C=&\left(\frac{1}{N}\sum x_i z_i'\right)\left(\frac{1}{N}\sum z_i z_i'\right)^{-1}
                    \left(\frac{1}{N}\sum \hat u_i^2 z_i z_i'\right)
                    \left(\frac{1}{N}\sum z_i z_i'\right)^{-1}\left(\frac{1}{N}\sum z_i x_i'\right).
                \end{align*}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Eficiencia}
    \begin{itemize}
        \item  Si asumimos homoscedasticidad, es decir IV6 $\E(u^2|z)=\E(u^2)=\sigma^2$:
                \begin{align*}
                    &\Avar(\hat \beta_{2SLS})=\frac{1}{N}\sigma^2 B^{-1}.
                \end{align*}
        \item Bajo IV6, 2SLS es el estimador asint\'oticamente m\'as eficiente entre los estimadores IV que usan como instrumentos combinaciones lineales de $z$.
    \end{itemize}
\end{frame}

\section{Validez de los instrumentos}

\begin{frame}
    \frametitle{Validez de los instrumentos}
		La consistencia y la distribuci\'on asint\'otica del estimador dependen de la validez de los instrumentos.
        \begin{itemize}
           \item \term{Relevancia}: Los instrumentos est\'an (parcialmente) correlacionados con la variable end\'ogena.
           \item \term{Exogeneidad}: Los instrumentos no est\'an correlacionados con los inobservados de la ecuaci\'on de inter\'es.
        \end{itemize}
\end{frame}
\subsection{Relevancia de los instrumentos}
\begin{frame}
    \frametitle{Relevancia de los instrumentos}
    \begin{itemize}
        \item  Condici\'on de rango: $rango(\E(zx'))=K$.
		\item  Para \term{una variable end\'ogena y un instrumento}, podemos contrastar la condici\'on de rango con un test de significancia del instrumento en la primera etapa.
		\item Para \term{una variable end\'ogena y m\'ultiples  instrumentos}, podemos contrastar la condici\'on de rango con un test de significancia conjunta de los instrumentos en la primera etapa.
		\item Para \term{m\'as de una variable end\'ogena}, una forma intuitiva de verla es que instrumentos distintos sean significativos para distintas variables end\'ogenas en la primera etapa. Existen contrastes que eval\'uan si la matriz $\E(zx')$ tiene rango $K$.
		\item  Contraste de Wald (o F) en la primera etapa $H_0:$ los coeficientes de los instrumentos son cero, versus $H_1:$ al menos uno de los coeficientes de los instrumentos es distinto de cero.
    \end{itemize}
\end{frame}

% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Condici\'on de rango}
    \begin{itemize}
        \item  Podemos escribir los instrumentos de la siguiente forma
        \begin{align*}
            x^* &=\Pi'z,
        \end{align*}
        donde $\Pi= \E(zz')^{-1}\E(zx')$
        \item  Recuerde que $\Pi$ es una matriz de $L \times K$ donde cada columna corresponde a la primera etapa de cada variable explicativa.
        \item  Podemos contrastar la condici\'on de rango con la matriz $\Pi$.
        \uncover<2-> {
        Demostraci\'on:
                \begin{align*}
                    \E(zx')&=\E(z(x^*+e)')=\E(zx^{*'})= \E(zz') \Pi.
                \end{align*}
                Dado que $rango(\E(zz'))=L$, necesitamos $rango(\Pi)=K$.
        }
    \end{itemize}
\end{frame}

% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Ejemplo 1: Modelo simple}
    \begin{itemize}
        \item  Modelo: $y=\beta_0+\beta_1x_1+u$ con un instrumento $z_1$.
        Escribimos la primera etapa como $x_{1}=\pi_{10}+\pi_{11}z_1+e_1$. Entonces
                \begin{align*}
                    \Pi = \left[ 
                    \begin{array} {c c}
                        1 & \pi_{10}\\
                        0 & \pi_{11}
                    \end{array}
                    \right].
                \end{align*}
        La condici\'on de rango es $\pi_{11}\neq 0$.
      \end{itemize}
\end{frame}

% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Ejemplo 2: Regresi\'on m\'ultiple con una variable end\'ogena $x_K$}
    \begin{itemize}
         \item  Escribimos la primera etapa como $x_{K}=x_{-K}'\pi_{10}+z_1'\pi_{11}+e_K$, donde $z_1$ son los $L_1$ instrumentos excluidos, $\pi_{10}$ es $(K-1)\times 1$ y $\pi_{11}$ es $L_1\times 1$. Entonces
        \begin{align*}
            \Pi = \left[ 
            \begin{array} {c c}
                I_{K-1} & \pi_{10}\\
                0_{L_1,K-1} & \pi_{11}
            \end{array}
            \right].
        \end{align*}
        Se necesita por lo menos un elemento de $\pi_{11}$ distinto de cero.
    \end{itemize}
\end{frame}

\subsection{Exogeneidad de los instrumentos}
\begin{frame}
    \frametitle{Contraste de sobreidentificaci\'on (Sargan--Hansen)}
    \small
    \begin{itemize}
        \item <1->Queremos contrastar la condici\'on de exogeneidad:
        \begin{align*}
            H_0: \E(zu)=0,\\
            H_1: \E(zu)\neq 0.
        \end{align*}
        \item <2-> Idea: Podemos utilizar
        \begin{align*}
            \sqrt{N}\frac{1}{N}\sum z_i u_i \dto_{H_{0}} \N(0,\E(u^2zz')),
        \end{align*}
        para construir el test estad\'istico
        \begin{align*}
            J_N=N\left(\frac{1}{N}\sum z_i u_i\right)'\left(\frac{1}{N}\sum u_i^2z_iz_i'\right)^{-1}\left(\frac{1}{N}\sum z_i u_i\right)\dto_{H_0} \chi^2_L.
        \end{align*}
        \item <3->
       Problema: No observamos $u_i$ y debemos reemplazarlo por residuos. Si el modelo est\'a exactamente identificado no podemos realizar el contraste porque $\sum z_i \hat u_i=0$.
    \end{itemize}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Contraste de sobreidentificaci\'on: Sargan y Hansen}
    \small
    \begin{itemize}
        \item \term{Test de Sargan}: con homoscedasticidad (IV6), reemplazamos $u_i$ por los residuos de 2SLS $\hat u_i$ y $\E(u^2zz')$ por $\hat\sigma^2\frac{1}{N}\sum z_iz_i'$. Al estimar $\beta$ perdemos $K$ grados de libertad:
        \begin{align*}
            S_N=N\left(\frac{1}{N}\sum z_i \hat u_i\right)'\left(\hat\sigma^2\frac{1}{N}\sum z_iz_i'\right)^{-1}\left(\frac{1}{N}\sum z_i \hat u_i\right)\dto_{H_0} \chi^2_{L-K}.
        \end{align*}
        \item <2-> \term{Test $J$ de Hansen} (robusto a heteroscedasticidad): usa $\frac{1}{N}\sum \hat u_i^2z_iz_i'$, pero con los residuos de GMM eficiente, el estimador que pondera los momentos con la inversa de esa matriz.
        \item <2-> Con los residuos de 2SLS y la ponderaci\'on robusta, el estad\'istico no tiene distribuci\'on $\chi^2_{L-K}$: la ponderaci\'on del contraste tiene que coincidir con la del estimador.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Discusi\'on}
    \small
    \begin{enumerate} [<+->]
        \item El estad\'istico de Sargan es igual a $LM_N=N\hat R^2\dto_{H_0} \chi^2_{L-K}$, donde $\hat R^2$ es el $R^2$ de la regresi\'on de $\hat u_i$ sobre $z_i$ (que incluye las ex\'ogenas incluidas).

        \item  El test de sobreidentificaci\'on no suele ser muy \'util en la pr\'actica porque los s.e. son muy grandes y es dif\'icil rechazar $H_0$. El test tiene poco poder.

        \item En R, \texttt{summary(iv, diagnostics = TRUE)} (paquete \texttt{ivreg}) reporta el test de Sargan, con o sin errores est\'andar robustos.
    \end{enumerate}
\end{frame}

\subsection{Contraste de endogeneidad}
\begin{frame}
    \frametitle{Contraste de endogeneidad (Durbin--Wu--Hausman)}
    \vspace*{-0.6em}
    \small
    \begin{itemize}
        \item <1->Queremos contrastar si la variable $x_K$ es end\'ogena:
        \begin{align*}
            H_0: \E(x_K u)=0,\\
            H_1: \E(x_K u)\neq 0.
        \end{align*}
        \item <2-> Idea (Hausman, 1978): comparar MCO, consistente y eficiente bajo $H_0$ (con homoscedasticidad), con IV, consistente bajo $H_0$ y $H_1$:
        \scriptsize{
        \begin{align*}
             \sqrt{N}(\hat \beta_{IV}-\hat \beta_{MCO}) \dto_{H_0} \N(0,\Avar(\sqrt{N}(\hat \beta_{IV}-\beta))-\Avar(\sqrt{N}(\hat \beta_{MCO}-\beta))),
        \end{align*}
        para construir el test estad\'istico
        \begin{align*}
            H_N= (\hat \beta_{1,IV}-\hat \beta_{1,MCO})'(\Avar(\hat \beta_{1,IV})-\Avar(\hat \beta_{1,MCO}))^{-1} (\hat \beta_{1,IV}-\hat \beta_{1,MCO})\dto_{H_0} \chi^2_{K_2},
        \end{align*}
        donde $\hat \beta_{1}$ son los coeficientes de las $K_2$ variables end\'ogenas.
        }
        \item <3-> En R, \texttt{summary(iv, vcov. = function(x) vcovHC(x, type = "HC0"), diagnostics = TRUE)} reporta este test (Wu--Hausman, paquete \texttt{ivreg}).
    \end{itemize}
\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
    \frametitle{Discusi\'on}
    \begin{enumerate} 
        \item <1-> En la pr\'actica, el test se hace con una \term{funci\'on de control}:
        \begin{enumerate}[(a)]
            \item Estimar la primera etapa por MCO y guardar los residuos,
            \item Estimar la ecuaci\'on de inter\'es por MCO controlando por los residuos obtenidos en (a),
            \item Hacer un contraste de Wald (robusto) sobre el coeficiente de los residuos.
        \end{enumerate}
        Con errores est\'andar robustos, esta versi\'on no requiere homoscedasticidad.
       	\item <2-> Es una forma alternativa de estimar los par\'ametros en forma consistente: enfoque de \term{funciones de control}. Una vez que controlamos por los residuos de la primera etapa, las variables explicativas no est\'an correlacionadas con los inobservados y podemos estimar por MCO.
    \end{enumerate}
\end{frame}

	\frametitle{Test de endogeneidad a mano}
	(c) Contraste de Wald (robusto) sobre el coeficiente de $\hat v_t$:
	\begin{align*}
		t=0.422/0.394=1.07, \qquad F(1,90)=t^2=1.15, \qquad \text{p-valor}=0.287.
	\end{align*}

\end{frame}
% --------------------------------------------------- Slide --
\begin{frame}
	\frametitle{Test de endogeneidad: advertencias}
	\begin{itemize}
		\item \term{Poco poder}, sobre todo con instrumentos d\'ebiles: no rechazar $H_0$ no implica que $x$ sea ex\'ogena. En Graddy, MCO da $-0.525$ y 2SLS $-0.821$, una diferencia grande, pero el test no rechaza (p-valor $=0.287$).
		\item La versi\'on cl\'asica de Hausman supone homoscedasticidad (MCO eficiente bajo $H_0$); la versi\'on por funci\'on de control con errores robustos no.
		\item Si el efecto de $x$ es heterog\'eneo entre individuos, MCO e IV pueden estimar par\'ametros distintos: una diferencia entre ambos no implica necesariamente endogeneidad.
		\item En la pr\'actica: reportar MCO e IV lado a lado y discutir la diferencia.
	\end{itemize}
\end{frame}
